openai

edge-tts
PyQt5
Pillow  # shared icon pipeline and Windows ICO generation
psutil

pytest

# IANA timezone database used by zoneinfo (Asia/Dhaka in router.py). Frozen
# Windows/macOS builds have no system tz database, so this must be bundled.
tzdata

# Offline speech-to-text fallback. If Google's online STT is unreachable,
# Jarvis transcribes locally instead (see voice._local_transcribe).
faster-whisper

# Semantic response cache and semantic-memory embeddings (all-MiniLM-L6-v2 via
# ONNX). Loaded lazily; if it is missing or the model cannot be downloaded,
# Jarvis falls back to exact-match caching / SQLite fact retrieval and all other
# features keep working.
fastembed

# Cross-platform audio capture and window control for Windows/macOS.
# On Linux Jarvis uses arecord/wmctrl/xdotool instead, but installing these is
# harmless there (the adapters import them lazily and degrade gracefully).
sounddevice
pywinctl

# Silero VAD endpointing. Loaded lazily by voice.py; if it is missing or fails
# to load, the calibrated RMS gate remains the fallback. ONNX keeps inference
# on CPU without requiring a CUDA build.
silero-vad
onnxruntime

# Persistent semantic memory (ChromaDB + the fastembed model listed above).
# ChromaDB is initialized lazily at ~/.jarvis/chroma_db and SQLite remains the
# fallback if it cannot be loaded.
chromadb

# Global wake hotkey (Ctrl+Shift+J) for the GUI. Imported lazily, so the app
# still starts if it is missing -- the hotkey simply is not available.
pynput

# Official Model Context Protocol SDK. Imported lazily by mcp_client.py, so
# Jarvis still runs without it; install this to use MCP_SERVERS in .env.
mcp
